# Run Upstreamed locally
# Open http://127.0.0.1:8000/blogs/ after starting. Press Ctrl+C to stop.

# First time only: install dependencies
cd ~/blogs && python3 -m venv .venv && .venv/bin/pip install -r requirements.txt

# Preview (auto-reloads when you save prs.yml)
cd ~/blogs && .venv/bin/mkdocs serve

# Add PRs: edit prs.yml, or run `make prs-new` to add the missing ones from GitHub

# Publish to GitHub Pages
cd ~/blogs && git add . && git commit -m "Update PRs" && git push
